% -----------------------------------------------------------------------------
% TikZ Externalization (Caching)
% -----------------------------------------------------------------------------
% How it works:
%   Each `tikzpicture` is compiled once into a standalone PDF in `tikz-cache/`.
%   Subsequent compilations simply include the pre-rendered PDF rather than
%   recalculating the entire diagram, drastically speeding up compilation.
%
% Requirements:
%   - The LaTeX compiler must run with `-shell-escape` (configured in VSCode
%     LaTeX-Workshop settings via `latex-workshop.latex.tools`).
%   - The target directory (`tikz-cache/`) must exist.
%
% How to bypass caching during active development (WIP figures):
%   - For a SINGLE figure (compile inline without caching):
%       Place `\tikzset{external/export next=false}` immediately before
%       `\begin{tikzpicture}`.
%   - For a whole FILE / BLOCK:
%       `\tikzset{external/export=false}` at start, `\tikzset{external/export=true}` at end.
%
% How to rebuild / invalidate cache:
%   - Rebuild ALL figures:
%       1. Delete cache files manually (e.g. `rm -f tikz-cache/*`), OR
%       2. Temporarily add: `\tikzset{external/force remake=true}`
%   - Rebuild ONLY ONE specific figure:
%       Place `\tikzset{external/remake next}` immediately before the target
%       `\begin{tikzpicture}` environment.
% -----------------------------------------------------------------------------
\tikzexternalize[prefix=tikz-cache/, only named]
