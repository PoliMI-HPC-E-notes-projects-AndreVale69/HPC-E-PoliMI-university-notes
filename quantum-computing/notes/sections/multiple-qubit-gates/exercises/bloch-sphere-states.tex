\subsubsection{Bloch sphere states}\label{sec:ex-bloch-sphere-states}

In general, the single-qubit state can be expressed as (\autopageref{eq:canonical-parametrization}):
\begin{equation*}
    \ket{\psi} = \cos\left(\frac{\theta}{2}\right)\ket{0} + e^{i\phi}\sin\left(\frac{\theta}{2}\right)\ket{1}
\end{equation*}
With:
\begin{equation*}
    \theta \in \left[0, \pi\right], \quad \phi \in \left[0, 2\pi\right)
\end{equation*}
And it identifies the main states of the Bloch sphere representation, where $\theta$ is the polar angle and $\phi$ is the azimuthal angle.

\highspace
\exercise Identify the states of the Bloch sphere representation corresponding to the following pairs of angles $(\theta, \phi)$:
\begin{equation*}
    \theta = 0, \quad \phi = 0
\end{equation*}
\solution The state corresponding to $\theta = 0$ and $\phi = 0$ is calculated substituting the values into the general expression for $\ket{\psi}$:
\begin{equation*}
    \begin{aligned}
        \ket{\psi} &= \cos\left(\frac{0}{2}\right)\ket{0} + e^{i \cdot 0}\sin\left(\frac{0}{2}\right)\ket{1} \\
        &= \cos(0)\ket{0} + e^{0i}\sin(0)\ket{1} \\
        &= 1\ket{0} + 1 \cdot 0\ket{1} \\
        &= \ket{0}
    \end{aligned}
\end{equation*}
So, the state corresponding to $\theta = 0$ and $\phi = 0$ is $\ket{0}$, which is the north pole of the Bloch sphere.

\highspace
\exercise Identify the states of the Bloch sphere representation corresponding to the following pairs of angles $(\theta, \phi)$:
\begin{equation*}
    \theta = \pi, \quad \phi = 0
\end{equation*}
\solution The state corresponding to $\theta = \pi$ and $\phi = 0$ is calculated substituting the values into the general expression for $\ket{\psi}$:
\begin{equation*}
    \begin{aligned}
        \ket{\psi} &= \cos\left(\frac{\pi}{2}\right)\ket{0} + e^{i \cdot 0}\sin\left(\frac{\pi}{2}\right)\ket{1} \\
        &= \cos\left(\frac{\pi}{2}\right)\ket{0} + e^{0i}\sin\left(\frac{\pi}{2}\right)\ket{1} \\
        &= 0\ket{0} + 1\ket{1} \\
        &= \ket{1}
    \end{aligned}
\end{equation*}
So, the state corresponding to $\theta = \pi$ and $\phi = 0$ is $\ket{1}$, which is the south pole of the Bloch sphere.

\highspace
\exercise Identify the states of the Bloch sphere representation corresponding to the following pairs of angles $(\theta, \phi)$:
\begin{equation*}
    \theta = \frac{\pi}{2}, \quad \phi = 0
\end{equation*}
\solution The state corresponding to $\theta = \frac{\pi}{2}$ and $\phi = 0$ is calculated substituting the values into the general expression for $\ket{\psi}$:
\begin{equation*}
    \begin{aligned}
        \ket{\psi} &= \cos\left(\frac{\pi}{4}\right)\ket{0} + e^{i \cdot 0}\sin\left(\frac{\pi}{4}\right)\ket{1} \\
        &= \frac{1}{\sqrt{2}}\ket{0} + e^{0i}\frac{1}{\sqrt{2}}\ket{1} \\
        &= \frac{1}{\sqrt{2}}\ket{0} + \frac{1}{\sqrt{2}}\ket{1} \\
        &= \frac{1}{\sqrt{2}}\left(\ket{0} + \ket{1}\right) \\
        &= \ket{+}
    \end{aligned}
\end{equation*}
So, the state corresponding to $\theta = \frac{\pi}{2}$ and $\phi = 0$ is $\ket{+}$, which is located on the equator of the Bloch sphere along the positive $x$-axis.

\highspace
\exercise Identify the states of the Bloch sphere representation corresponding to the following pairs of angles $(\theta, \phi)$:
\begin{equation*}
    \theta = \frac{\pi}{2}, \quad \phi = \pi
\end{equation*}
\solution The state corresponding to $\theta = \frac{\pi}{2}$ and $\phi = \pi$ is calculated substituting the values into the general expression for $\ket{\psi}$:
\begin{equation*}
    \begin{aligned}
        \ket{\psi} &= \cos\left(\frac{\pi}{4}\right)\ket{0} + e^{i \cdot \pi}\sin\left(\frac{\pi}{4}\right)\ket{1} \\
        &= \frac{1}{\sqrt{2}}\ket{0} + e^{i\pi}\frac{1}{\sqrt{2}}\ket{1} \\
        &= \frac{1}{\sqrt{2}}\ket{0} - \frac{1}{\sqrt{2}}\ket{1} \\
        &= \frac{1}{\sqrt{2}}\left(\ket{0} - \ket{1}\right) \\
        &= \ket{-}
    \end{aligned}
\end{equation*}
The exponential term $e^{i\pi}$ evaluates to $-1$ due to Euler's formula. Recall the Euler's formula for any angle $\theta$:
\begin{equation}\label{eq:euler-formula}
    e^{i\theta} = \cos(\theta) + i\sin(\theta)
\end{equation}
If we substitute $\theta = \pi$ into \autoref{eq:euler-formula}, we get:
\begin{equation*}
    e^{i\pi} = \cos(\pi) + i\sin(\pi) = -1 + 0i = -1
\end{equation*}
So, the state corresponding to $\theta = \frac{\pi}{2}$ and $\phi = \pi$ is $\ket{-}$, which is located on the equator of the Bloch sphere along the negative $x$-axis.
